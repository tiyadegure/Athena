### 2.2 前端改造（frontend/app.js）

#### 改动 1：加载 report.json

```javascript
// 替换 SAMPLE_REPORT 硬编码
async function loadReport() {
    try {
        const response = await fetch('report.json');
        if (!response.ok) throw new Error('Failed to load report');
        return await response.json();
    } catch (e) {
        console.warn('Using embedded sample data');
        return SAMPLE_REPORT;  // fallback
    }
}
```

#### 改动 2：MetaMask 连接

```javascript
let connectedAccount = null;

async function connectWallet() {
    if (!window.ethereum) {
        alert('请安装 MetaMask');
        return;
    }
    const accounts = await window.ethereum.request({
        method: 'eth_requestAccounts'
    });
    connectedAccount = accounts[0];
    // 更新 UI 显示地址
    updateWalletUI(connectedAccount);
}

// 检测链是否为 Sepolia
async function checkNetwork() {
    const chainId = await window.ethereum.request({ method: 'eth_chainId' });
    if (chainId !== '0xaa36a7') {  // 11155111 = Sepolia
        await window.ethereum.request({
            method: 'wallet_switchEthereumChain',
            params: [{ chainId: '0xaa36a7' }]
        });
    }
}
```

#### 改动 3：NFT 铸造

```javascript
const NFT_ABI = [
    "function mint(address to, uint256 tier, bytes32 attestationUID) external",
    "function balanceOf(address account, uint256 id) external view returns (uint256)"
];

async function mintCertificate(report) {
    if (!connectedAccount) {
        alert('请先连接钱包');
        return;
    }

    const tierMap = { 'Gold': 1, 'Silver': 2, 'Bronze': 3 };
    const tier = tierMap[report.certificate.tier] || 3;
    const attestationUID = report.attestation.uid;

    const contract = new ethers.Contract(NFT_CONTRACT, NFT_ABI, signer);
    const tx = await contract.mint(connectedAccount, tier, attestationUID);
    
    showMintStatus('pending', tx.hash);
    const receipt = await tx.wait();
    showMintStatus('success', receipt.transactionHash);
}
```

#### 改动 4：渲染报告数据

需要新增的渲染函数：
- `renderSummary(data)` — 审计概览（漏洞数、评分、合约数）
- `renderFindings(data.findings)` — 漏洞列表（严重性排序、展开详情）
- `renderVerification(data.verification)` — 验证引擎状态
- `renderAttackSimulation(data.attack_simulation)` — 攻击模拟展示
- `renderAttestation(data.attestation)` — EAS 链上认证 + 链接
- `renderCertificate(data.certificate)` — NFT traits 展示 + Mint 按钮

### 2.5 ethers.js 依赖

前端需要 ethers.js 来与合约交互。两个选项：

**选项 A：CDN 引入（推荐，简单）**
```html
<script src="https://cdn.ethers.io/lib/ethers-5.7.2.umd.min.js"></script>
```

**选项 B：npm install**
```bash
npm install ethers
```

用选项 A，不需要构建步骤。
